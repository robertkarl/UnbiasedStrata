cmake_minimum_required(VERSION 3.24)
project(strata_muse LANGUAGES C CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_RUNTIME_LIBRARY Shared)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
get_filename_component(STRATA_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(GGML_CUDA ON CACHE BOOL "" FORCE)
foreach(opt LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_TOOLS LLAMA_BUILD_SERVER
            LLAMA_BUILD_APP LLAMA_OPENSSL LLAMA_LLGUIDANCE LLAMA_USE_PREBUILT_UI)
  set(${opt} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(${STRATA_ROOT}/third_party/llama-muse llama-muse EXCLUDE_FROM_ALL)
find_package(CUDAToolkit REQUIRED)
add_library(strata_muse STATIC
  ${STRATA_ROOT}/src/muse/model.cpp
  ${STRATA_ROOT}/src/kernels/cuda/muse.cu
  ${STRATA_ROOT}/src/kernels/cuda/elementwise.cu
  ${STRATA_ROOT}/src/kernels/cuda/iq_kernels.cu)
target_include_directories(strata_muse PUBLIC ${STRATA_ROOT}/include ${STRATA_ROOT}/src/muse
  PRIVATE ${STRATA_ROOT}/third_party/llama.cpp/ggml/src)
target_link_libraries(strata_muse PUBLIC CUDA::cudart CUDA::cublas ggml-base)
target_compile_options(strata_muse PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--use_fast_math>)
add_executable(muse-check ${STRATA_ROOT}/tools/muse_check.cpp)
target_link_libraries(muse-check PRIVATE strata_muse llama)
enable_testing()
add_executable(muse-kernel-test ${STRATA_ROOT}/tests/core/muse_kernels_test.cpp)
target_link_libraries(muse-kernel-test PRIVATE strata_muse)
add_test(NAME muse-kernels COMMAND muse-kernel-test)
find_package(Threads REQUIRED)
add_executable(unbiased-strata ${STRATA_ROOT}/frontend/main.cpp ${STRATA_ROOT}/src/muse/serve.cpp)
target_compile_definitions(unbiased-strata PRIVATE STRATA_MUSE_NATIVE=1)
target_include_directories(unbiased-strata PRIVATE ${STRATA_ROOT}/third_party/llama-muse/vendor)
target_link_libraries(unbiased-strata PRIVATE strata_muse llama-common llama mtmd cpp-httplib Threads::Threads)
add_executable(muse-chat-test ${STRATA_ROOT}/frontend/muse_chat_parser_test.cpp)
target_compile_definitions(muse-chat-test PRIVATE STRATA_MUSE_NATIVE=1)
target_include_directories(muse-chat-test PRIVATE ${STRATA_ROOT}/third_party/llama-muse/vendor)
target_link_libraries(muse-chat-test PRIVATE llama-common llama mtmd cpp-httplib Threads::Threads)
