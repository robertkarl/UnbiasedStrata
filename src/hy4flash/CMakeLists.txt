# Standalone so building this model does not configure unrelated engines.
cmake_minimum_required(VERSION 3.24)
project(strata_hy4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(strata_root "../.." ABSOLUTE)
set(reader ../artifact/strata_reader.cpp run/hy4_engram.cpp)
find_path(ZSTD_INCLUDE_DIR zstd.h REQUIRED)
find_library(ZSTD_LIBRARY zstd REQUIRED)
find_package(Threads REQUIRED)
if(APPLE)
    enable_language(OBJCXX)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    file(GLOB kernels CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/metal/*.metal")
    set(embedded "${CMAKE_CURRENT_BINARY_DIR}/hy4_metal_embedded.inc")
    add_custom_command(OUTPUT "${embedded}"
        COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/metal/embed_metal.py" "${embedded}" ${kernels}
        DEPENDS metal/embed_metal.py ${kernels} VERBATIM)
    add_executable(strata-hy4 metal/hy4_metal_main.cpp metal/mtl_rt.mm ${embedded} ${reader})
    target_compile_definitions(strata-hy4 PRIVATE HY4_EMBED_KERNELS STRATA_GGML_COMMON_DEFAULT="" STRATA_METAL_KERNELS_DEFAULT="")
    target_include_directories(strata-hy4 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_options(strata-hy4 PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
    target_link_libraries(strata-hy4 PRIVATE "-framework Metal" "-framework Foundation")
else()
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    set(gpu_sources run/hy4_main.hip.cpp run/mla_attn.hip.cpp)
    set_source_files_properties(${gpu_sources} PROPERTIES LANGUAGE CUDA)
    add_executable(strata-hy4 ${gpu_sources} run/hy4_cpu_experts.cpp ../quant/sk_codecs.cpp ${reader})
    target_compile_options(strata-hy4 PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
    set_target_properties(strata-hy4 PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_link_libraries(strata-hy4 PRIVATE CUDA::cublas)
endif()
target_include_directories(strata-hy4 PRIVATE run "${strata_root}/include" "${strata_root}/third_party/llama.cpp/vendor/stb" "${ZSTD_INCLUDE_DIR}")
target_link_libraries(strata-hy4 PRIVATE "${ZSTD_LIBRARY}" Threads::Threads)
